@app.route("/")

def index(): return HTML

@app.route("/stream")
@auth.login_required
def stream():
    def gen():
        with requests.get(f"http://{CAM_ESP}/stream", stream=True, timeout=10) as r:
            for chunk in r.iter_content(1024): yield chunk
    return Response(gen(), mimetype="multipart/x-mixed-replace; boundary=frame")

@app.route("/move")
@auth.login_required
def manual_move():
    dir = request.args.get("dir", "stop")
    with move_lock:
        global current_move
        if dir == "stop":
            current_move = None
            send_cmd("x")
        elif dir in "wasd":
            current_move = dir
    return "OK"

@app.route("/cmd")
@auth.login_required
def single_cmd():
    c = request.args.get("c", "")
    if c == "x":
        with move_lock:
            global current_move, ai_move_target
            current_move = None
            ai_move_target = None
        send_cmd("x")
    elif c == "l":
        send_cmd("l")
    return "OK"

@app.route("/toggle")
@auth.login_required
def toggle():
    global ai_active
    ai_active = not ai_active
    if ai_active:
        speak("Hello friend! I'm awake and ready!")
    return jsonify(ai_active)

@app.route("/speed")
@auth.login_required
def setspeed():
    global throttle
    try:
        s = int(request.args.get("s", 13))
        throttle = max(5, min(13, s))
        set_throttle(throttle)
        log(f"Throttle set to {throttle}")
    except:
        pass
    return "OK"
