void setup() {
  pinMode(L_PWM, OUTPUT); pinMode(L_DIR, OUTPUT);
  pinMode(R_PWM, OUTPUT); pinMode(R_DIR, OUTPUT);
  pinMode(LIGHT_PIN, OUTPUT);
  digitalWrite(LIGHT_PIN, LOW);    // start OFF

  stopMotors();
  Serial.begin(115200);

  WiFi.config(local_IP, gateway, subnet);
  WiFi.begin(ssid, password);
  while(WiFi.status()!=WL_CONNECTED) delay(500);

  Serial.print("IP: http://"); Serial.println(WiFi.localIP());

  server.on("/", [](){ server.send_P(200, "text/html", page); });
  server.on("/c", handleCmd);
  server.begin();
}

void loop() {
  server.handleClient();
  if(cmd!='0' && millis()-lastCmd>300) stopMotors();
}

void stopMotors() {
  analogWrite(L_PWM,0); analogWrite(R_PWM,0); cmd='0';
}

void drive(int l, int r) {
  digitalWrite(L_DIR, l<0?HIGH:LOW); analogWrite(L_PWM, abs(l));
  digitalWrite(R_DIR, r<0?LOW:HIGH); analogWrite(R_PWM, abs(r));
}

void handleCmd() {
  if(!server.hasArg("d")) return;
  String v = server.arg("d");
  lastCmd = millis();

  if(v=="w") { drive(currentSpeed, currentSpeed); cmd='w'; }
  else if(v=="s") { drive(-currentSpeed, -currentSpeed); cmd='s'; }
  else if(v=="a") { drive(-currentSpeed, currentSpeed); cmd='a'; }
  else if(v=="d") { drive(currentSpeed, -currentSpeed); cmd='d'; }
  else if(v=="x") { stopMotors(); cmd='0'; }
  else if(v.startsWith("s")) {
    int newSpeed = v.substring(1).toInt();
    if(newSpeed>=5 && newSpeed<=13) currentSpeed = newSpeed;
  }
  
  // LIGHT TOGGLE COMMAND
  else if(v == "l") {
    lightState = !lightState;
    digitalWrite(LIGHT_PIN, lightState ? HIGH : LOW);   // 3.3V HIGH = ON → perfect for MOSFET gate
  }

  server.send(200, "text/plain", "OK");
}
